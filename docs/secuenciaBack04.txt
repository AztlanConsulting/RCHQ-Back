@startuml
title US04 back - user cambia contraseña

actor Usuario
boundary "user.route.js" as Route
control "verifyToken" as Auth
control "validate(changePasswordSchema)" as Validate
control "user.controller.js" as Controller
control "password.service.js" as Service
entity "user.model.js" as UserModel
collections "password utils" as PasswordUtil
entity "log.model.js" as LogModel
database "BD" as DB

Usuario -> Route : POST /change-password\ncurrentPassword, newPassword, confirmPassword
Route -> Auth : Validar JWT

alt Token inválido
Auth --> Usuario : 401 Unauthorized
else Token válido
Auth -> Validate : Validar request

alt Datos inválidos
Validate --> Usuario : 400 Error de validación
else Datos válidos
Validate -> Controller : changePassword(req)
Controller -> Service : changePassword(req)

Service -> UserModel : getEmployeeById(req.user.id)
UserModel -> DB : Consultar empleado
DB --> UserModel : Empleado / no encontrado
UserModel --> Service : Empleado / no encontrado

alt Usuario no existe
Service --> Controller : 404 Employee not found
Controller --> Usuario : 404 Employee not found
else Usuario existe
alt Cuenta inactiva
Service -> LogModel : Guardar intento con cuenta inactiva
LogModel -> DB : Insertar log
DB --> LogModel : Confirmación
Service --> Controller : 403 Access not allowed
Controller --> Usuario : 403 Access not allowed
else Cuenta activa
alt Confirmación no coincide o formato inválido
Service --> Controller : 400 Error de validación
Controller --> Usuario : 400 Error
else Datos válidos
Service -> PasswordUtil : verifyPassword(currentPassword, employee.pwd)
PasswordUtil --> Service : true / false

alt Contraseña actual incorrecta
Service -> LogModel : Guardar intento fallido de cambio
LogModel -> DB : Insertar log
DB --> LogModel : Confirmación
Service --> Controller : 401 Invalid credentials
Controller --> Usuario : 401 Invalid credentials
else Contraseña actual correcta
Service -> PasswordUtil : verifyPassword(newPassword, employee.pwd)
PasswordUtil --> Service : true / false

alt Nueva igual a la actual
Service --> Controller : 400 New password must be different
Controller --> Usuario : 400 Error
else Nueva distinta
Service -> PasswordUtil : hashPassword(newPassword)
PasswordUtil --> Service : hashedPassword

Service -> DB : Actualizar password
DB --> Service : Confirmación

Service -> LogModel : Guardar PASSWORD_CHANGED_SUCCESS
LogModel -> DB : Insertar log
DB --> LogModel : Confirmación

Service --> Controller : 200 Password changed successfully
Controller --> Usuario : 200 Éxito
end
end
end
end
end
end
end

@enduml
